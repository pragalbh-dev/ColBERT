# EXPERIMENT MANAGER

## Purpose
Organize related training jobs and provide comparison utilities.

## Responsibilities
1. Create and track experiments
2. Create training jobs within experiments
3. Compare results across training jobs
4. Visualize metrics via TensorBoard
5. Manage experiment artifacts and configurations
6. Orchestrate multi-phase training workflows
7. Manage test datasets and evaluations

## Responsibility Clarification
The Experiment Manager is the ONLY component responsible for:
- Creating and organizing experiments
- Managing collections of related training jobs
- Providing comparison utilities across jobs
- Serving as the main entry point for the experiment workflow

The Experiment Manager delegates individual job execution to Training Job instances,
configuration management to the Config Manager, and artifact management to the
Artifact Manager.

    def create_evaluation_job(self, 
                             job_id: str,
                             evaluation_script: str,
                             model_path: str,
                             dataset_path: str,
                             config: Config) -> EvaluationJob:
        """
        Create an evaluation job
        
        Args:
            job_id: Unique identifier for the job
            evaluation_script: Path to evaluation script
            model_path: Path to model to evaluate
            dataset_path: Path to test dataset
            config: Configuration for the job
            
        Returns:
            EvaluationJob instance
        """
        # Create evaluation job
        job = EvaluationJob(
            job_id=job_id,
            experiment_name=self.name,
            evaluation_script=evaluation_script,
            model_path=model_path,
            dataset_path=dataset_path,
            config=config,
            artifact_manager=self.artifact_manager
        )
        
        # Register job with experiment
        self._evaluation_jobs[job_id] = job
        
        return job
    
    def register_dataset(self, 
                        dataset_name: str, 
                        dataset_path: str, 
                        dataset_type: str = "test") -> str:
        """
        Register a dataset with the experiment
        
        Args:
            dataset_name: Name of the dataset
            dataset_path: Path to the dataset
            dataset_type: Type of dataset (train, test, validation)
            
        Returns:
            Path to the registered dataset
        """
        # Register dataset with artifact manager
        return self.artifact_manager.register_dataset(
            dataset_path=dataset_path,
            dataset_name=f"{self.name}_{dataset_type}_{dataset_name}",
            version="latest"
        )
    
    def compare_models(self, model_configs: List[Dict[str, str]]) -> Dict[str, Any]:
        """
        Compare multiple models based on their evaluation results
        
        Args:
            model_configs: List of dictionaries with model information
                Each dict should have: name, job_id, eval_job_id
                
        Returns:
            Comparison results
        """

            name = config["name"]
            training_job_id = config["job_id"]
            eval_job_id = config["eval_job_id"]

            # Get evaluation job
            eval_job = self.get_evaluation_job(eval_job_id)

            # Add to comparison

                "training_config": training_job.get_config().to_dict(),
                "evaluation_results": eval_results,
                "model_path": training_job.get_artifacts().get("model")
            }
        
        return results
