from experiment_module import ExperimentManager
from experiment_module.config import ConfigManager
import time

# 1. Create experiment
experiment = ExperimentManager.create("colbert_base_model_comparison")

# 2. Create base configuration
base_config = ConfigManager.create({
    # ColBERT specific parameters
    "dim": 128,
    "similarity": "cosine",
    "mask_punctuation": True,
    "doc_maxlen": 180,
    "query_maxlen": 32,
    "nway": 2,
    "use_ib_negatives": False,
    
    # Training parameters
    "batch_size": 32,
    "learning_rate": 3e-6,
    "warmup_steps": 1000,
    "max_steps": 400000,
    "gradient_accumulation_steps": 1,
    
    # Optimizer parameters
    "optimizer": "AdamW",
    "weight_decay": 0.01,
    
    # Dataset parameters
    "train_dataset": "msmarco",
    "eval_dataset": "msmarco-dev",
    
    # Checkpoint parameters
    "save_every": 10000,
    "eval_every": 10000,
})

# 3. Create jobs for different base models
base_models = [
    "bert-base-uncased",
    "bert-large-uncased",
    "roberta-base",
    "microsoft/mdeberta-v3-base",
    "xlm-roberta-base"
]

jobs = []
for base_model in base_models:
    # Create a config variant for this specific base model
    model_config = ConfigManager.create_from(base_config)
    model_config.update({
        "base_model": base_model,
        "batch_size": 16 if "large" in base_model else 32,
    })
    
    # Create a job with a descriptive ID
    job_id = f"colbert_{base_model.replace('/', '_')}"
    job = experiment.create_job(
        job_id=job_id,
        training_script="training_scripts/train_colbert.py",
        config=model_config,
        dataset_path="data/msmarco",
        framework="pytorch"
    )
    
    jobs.append(job)

# 4. Run jobs sequentially
for i, job in enumerate(jobs):
    print(f"Running job {i+1}/{len(jobs)}: {job.id}")
    results = job.run(gpu_id=0)
    print(f"Job completed with results: {results}")
    
    # Wait a bit between jobs to ensure clean GPU memory
    time.sleep(10)

# 5. Compare results
comparison = experiment.compare_jobs([job.id for job in jobs])

# Print comparison table
print("\nComparison of ColBERT models with different base models:")
print("-" * 80)
print(f"{'Base Model':<30} {'Final Loss':<15} {'MRR':<10} {'Recall@1':<10} {'Recall@5':<10} {'Recall@10':<10}")
print("-" * 80)

for job_id, metrics in comparison.items():
    base_model = metrics["config"]["base_model"]
    final_loss = metrics["results"]["final_loss"]
    eval_metrics = metrics["results"]["final_eval_metrics"]
    
    print(f"{base_model:<30} {final_loss:<15.4f} {eval_metrics['mrr']:<10.4f} "
          f"{eval_metrics['recall@1']:<10.4f} {eval_metrics['recall@5']:<10.4f} "
          f"{eval_metrics['recall@10']:<10.4f}")

# 6. Launch TensorBoard for visualization
tensorboard_process = experiment.launch_tensorboard(port=6006)
print("TensorBoard launched at http://localhost:6006")

# Keep the script running to keep TensorBoard alive
input("Press Enter to stop TensorBoard and exit...")
tensorboard_process.terminate()

best_job = experiment.get_job(best_job_id)
best_model_path = best_job.get_artifacts()["model"]

from experiment_module import ExperimentManager, ConfigManager
import time
import json

def run_multi_phase_colbert():
    # Create or load experiment
    experiment_name = "colbert_multi_phase"
    try:
        experiment = ExperimentManager.load(experiment_name)
        print(f"Loaded existing experiment: {experiment_name}")
    except:
        experiment = ExperimentManager.create(experiment_name)
        print(f"Created new experiment: {experiment_name}")
    
    # Create configurations for each phase
    phase1_config = ConfigManager.create("colbert_phase1", {
        "learning_rate": 3e-5,
        "batch_size": 32,
        "epochs": 10,
        "model_type": "bert-base-uncased",
        "max_seq_length": 128
    })
    
    phase2_config = ConfigManager.create("colbert_phase2", {
        "learning_rate": 1e-5,
        "batch_size": 32,
        "epochs": 5,
        "use_hard_negatives": True,
        "max_seq_length": 128
    })
    
    phase3_config = ConfigManager.create("colbert_phase3", {
        "learning_rate": 5e-6,
        "batch_size": 32,
        "epochs": 3,
        "distillation": True,
        "teacher_temperature": 2.0,
        "max_seq_length": 128
    })
    
    # Register test datasets
    test_dataset1 = experiment.register_dataset("test_data_phase1", "data/test_basic.json")
    test_dataset2 = experiment.register_dataset("test_data_phase2", "data/test_hard.json")
    test_dataset3 = experiment.register_dataset("test_data_phase3", "data/test_comprehensive.json")
    
    # Track results for final comparison
    results = {}
    
    # Phase 1: Initial Training
    print("\n=== Starting Phase 1: Initial Training ===")
    timestamp = int(time.time())
    phase1_job_id = f"phase1_training_{timestamp}"
    
    phase1_job = experiment.create_training_job(
        job_id=phase1_job_id,
        training_script="train_colbert.py",
        config=phase1_config,
        dataset_path="data/train_basic"
    )
    
    # Run the training job
    phase1_results = phase1_job.run(gpu_id=0)
    print(f"Phase 1 Training completed with results: {phase1_results}")
    
    # Get the trained model path
    phase1_artifacts = phase1_job.get_artifacts()
    phase1_model_path = phase1_artifacts["model"]
    
    # Phase 1 Evaluation
    print("\n=== Starting Phase 1: Evaluation ===")
    phase1_eval_job = experiment.create_evaluation_job(
        job_id=f"phase1_eval_{timestamp}",
        evaluation_script="evaluate_colbert.py",
        model_path=phase1_model_path,
        dataset_path=test_dataset1,
        config=phase1_config
    )
    
    # Run the evaluation
    phase1_eval_results = phase1_eval_job.run(gpu_id=0)
    print(f"Phase 1 Evaluation Results: {phase1_eval_results}")
    
    # Store results for comparison
    results["phase1"] = {
        "training_job_id": phase1_job_id,
        "model_path": phase1_model_path,
        "eval_results": phase1_eval_results
    }
    
    # Decision point - continue if performance is good enough
    if phase1_eval_results.get("recall@10", 0) < 0.7:
        print("Phase 1 performance below threshold, stopping experiment")
        save_results(experiment_name, results)
        return results
    
    # Phase 2: Fine-tuning
    print("\n=== Starting Phase 2: Fine-tuning ===")
    timestamp = int(time.time())
    phase2_job_id = f"phase2_training_{timestamp}"
    
    phase2_job = experiment.create_training_job(
        job_id=phase2_job_id,
        training_script="train_colbert.py",
        config=phase2_config,
        dataset_path="data/train_hard_negatives"
    )
    
    # Set pretrained model path in environment
    phase2_job.set_environment_variable("PRETRAINED_MODEL_PATH", phase1_model_path)
    
    # Run the training job
    phase2_results = phase2_job.run(gpu_id=0)
    print(f"Phase 2 Training completed with results: {phase2_results}")
    
    # Get the trained model path
    phase2_artifacts = phase2_job.get_artifacts()
    phase2_model_path = phase2_artifacts["model"]
    
    # Phase 2 Evaluation
    print("\n=== Starting Phase 2: Evaluation ===")
    phase2_eval_job = experiment.create_evaluation_job(
        job_id=f"phase2_eval_{timestamp}",
        evaluation_script="evaluate_colbert.py",
        model_path=phase2_model_path,
        dataset_path=test_dataset2,
        config=phase2_config
    )
    
    # Run the evaluation
    phase2_eval_results = phase2_eval_job.run(gpu_id=0)
    print(f"Phase 2 Evaluation Results: {phase2_eval_results}")
    
    # Store results for comparison
    results["phase2"] = {
        "training_job_id": phase2_job_id,
        "model_path": phase2_model_path,
        "eval_results": phase2_eval_results
    }
    
    # Decision point
    if phase2_eval_results.get("recall@10", 0) < 0.8:
        print("Phase 2 performance below threshold, stopping experiment")
        save_results(experiment_name, results)
        return results
    
    # Phase 3: Final Refinement
    print("\n=== Starting Phase 3: Final Refinement ===")
    timestamp = int(time.time())
    phase3_job_id = f"phase3_training_{timestamp}"
    
    phase3_job = experiment.create_training_job(
        job_id=phase3_job_id,
        training_script="train_colbert_distillation.py",  # Different script for distillation
        config=phase3_config,
        dataset_path="data/train_distillation"
    )
    
    # Set pretrained model path and teacher model in environment
    phase3_job.set_environment_variable("PRETRAINED_MODEL_PATH", phase2_model_path)
    phase3_job.set_environment_variable("TEACHER_MODEL_PATH", phase2_model_path)
    
    # Run the training job
    phase3_results = phase3_job.run(gpu_id=0)
    print(f"Phase 3 Training completed with results: {phase3_results}")
    
    # Get the trained model path
    phase3_artifacts = phase3_job.get_artifacts()
    phase3_model_path = phase3_artifacts["model"]
    
    # Final Comprehensive Evaluation
    print("\n=== Starting Final Comprehensive Evaluation ===")
    final_eval_job = experiment.create_evaluation_job(
        job_id=f"final_eval_{timestamp}",
        evaluation_script="evaluate_colbert_comprehensive.py",
        model_path=phase3_model_path,
        dataset_path=test_dataset3,
        config=phase3_config
    )
    
    # Run the evaluation
    final_eval_results = final_eval_job.run(gpu_id=0)
    print(f"Final Evaluation Results: {final_eval_results}")
    
    # Store results for comparison
    results["phase3"] = {
        "training_job_id": phase3_job_id,
        "model_path": phase3_model_path,
        "eval_results": final_eval_results
    }
    
    # Save all results
    save_results(experiment_name, results)
    
    # Generate comparison report
    generate_comparison_report(experiment, results)
    
    return results

def save_results(experiment_name, results):
    """Save multi-phase results to a file"""
    with open(f"{experiment_name}_results.json", "w") as f:
        json.dump(results, f, indent=2)

def generate_comparison_report(experiment, results):
    """Generate a simple comparison report"""
    # Create a list of models to compare
    models_to_compare = []
    
    for phase, phase_results in results.items():
        models_to_compare.append({
            "name": phase,
            "job_id": phase_results["training_job_id"],
            "eval_results": phase_results["eval_results"]
        })
    
    # Use experiment manager to compare models
    comparison = experiment.compare_models(models_to_compare)
    
    # Generate report
    experiment.generate_report(comparison, f"{experiment.name}_comparison_report.html")

if __name__ == "__main__":
    run_multi_phase_colbert()
