Basic usage
with BaseTracker("my_experiment", "job_123") as tracker:
# Log metrics
tracker.log_metric("accuracy", 0.85, step=1)
tracker.log_metrics({"loss": 0.1, "precision": 0.9}, step=1)
# Log artifacts
tracker.log_artifact("dataset", "path/to/dataset.csv")
# Add visualizations
tracker.add_text("description", "This is a test job")
# Add hyperparameters
tracker.add_hparams(
{"learning_rate": 0.01, "batch_size": 32},
{"accuracy": 0.85}
)
